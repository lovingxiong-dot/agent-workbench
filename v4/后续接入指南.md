### 文件：`v4/worker_manager.py`

### 需要适配的代码

```python
def _create_worker(self, event):
    """创建 Worker，接入现有 AgentWorker"""
    from workers.agent_worker import AgentWorker
    from agent_engine.llm_registry import LLMRegistry
    from services.config_service import ConfigService
    
    # 1. 解析 LLM（使用现有 LLMRegistry）
    llm_registry = LLMRegistry("config/config.yaml", "config/config.yaml")
    llm = llm_registry.get_llm(event.model)
    if not llm:
        llm = llm_registry.get_llm("tool-agent")  # 回退
    if not llm:
        self._bus.emit(WorkerErrorEvent(
            session_id=event.session_id,
            code="LLM_NOT_FOUND",
            detail=f"Model {event.model} not found",
        ))
        return
    
    # 2. 获取工具配置（从现有 config）
    config = ConfigService("config/config.yaml")
    mode_config = config.get("manual_modes", {}).get(event.mode, {})
    tools = mode_config.get("tools", [])
    system_prompt = self._build_system_prompt(event.mode, event.project_root)
    
    # 3. 创建 AgentWorker（使用现有类）
    worker = AgentWorker(
        mode_name=event.mode,
        current_llm=llm,
        current_tools=tools,
        session_id=event.session_id,
        system_prompt=system_prompt,
        tool_map=TOOL_MAP,  # 从现有 main_window.py 复制
        tool_definitions=TOOL_DEFINITIONS,  # 从现有 main_window.py 复制
        project_root=event.project_root,  # ← 环境绑定
        workspace_context="",
    )
    
    # 4. 存储并桥接信号
    self._workers[event.session_id] = worker
    
    # 桥接信号（与现有 main_window.py 中相同）
    worker.chunk_ready.connect(
        lambda chunk: self._bus.emit(WorkerChunkEvent(
            session_id=event.session_id,
            worker_id=worker.worker_id,
            chunk=chunk,
        ))
    )
    worker.result_ready.connect(
        lambda sid, text: self._bus.emit(WorkerResultEvent(
            session_id=event.session_id,
            worker_id=worker.worker_id,
            full_text=text,
        ))
    )
    worker.error_occurred.connect(
        lambda code, detail: self._bus.emit(WorkerErrorEvent(
            session_id=event.session_id,
            worker_id=worker.worker_id,
            code=code,
            detail=detail,
        ))
    )
    worker.tool_executed.connect(
        lambda name, args, result, elapsed: self._bus.emit(WorkerToolEvent(
            session_id=event.session_id,
            worker_id=worker.worker_id,
            tool_name=name,
            args=args,
            result=result,
        ))
    )
    
    # 5. 启动
    worker.start()
    self._bus.emit(WorkerCreatedEvent(
        session_id=event.session_id,
        worker_id=worker.worker_id,
    ))

def _build_system_prompt(self, mode, project_root):
    """构建 system prompt（接入现有逻辑）"""
    base = "你是一个专业的 AI 助手。"
    if project_root:
        base += f"\n当前工作目录: {project_root}"
    return base
```

### 文件：`v4/orchestrator.py`

```python
def _on_queue_task_ready(self, event):
    """队列任务就绪，启动 Phase 工作流"""
    rt = self._require_runtime(event.session_id)
    if not rt:
        return
    
    # 1. 创建 Worker（这会触发 WorkerManager 创建 AgentWorker）
    env = rt.environment
    self._bus.emit(WorkerCreateEvent(
        session_id=event.session_id,
        worker_id="",
        mode=rt.metadata.mode,
        model=rt.metadata.model,
        project_root=env.project_root,
    ))
    
    # 2. 获取或创建 PhaseManager（每个 SessionRuntime 一个）
    if not hasattr(rt, '_phase_manager'):
        from agent_engine.phase_manager import PhaseManager
        rt._phase_manager = PhaseManager(parent=self)
        
        # 连接 PhaseManager 信号到事件总线
        rt._phase_manager.phase_changed.connect(
            lambda phase, mode: self._bus.emit(PhaseChangedEvent(
                session_id=event.session_id,
                phase=phase,
                task_count=len(rt._phase_manager._context.task_list) if hasattr(rt._phase_manager, '_context') else 0,
            ))
        )
        rt._phase_manager.confirm_required.connect(
            lambda task_list: self._bus.emit(PhaseConfirmEvent(
                session_id=event.session_id,
                task_list=task_list,
            ))
        )
        rt._phase_manager.execute_required.connect(
            lambda task_list: self._bus.emit(PhaseExecuteEvent(
                session_id=event.session_id,
                task_list=task_list,
                original_text=rt.queue.get_streaming_task().user_text if rt.queue.get_streaming_task() else "",
            ))
        )
        rt._phase_manager.verify_required.connect(
            lambda results, mode: self._bus.emit(PhaseVerifyEvent(
                session_id=event.session_id,
                execution_results=results,
            ))
        )
        rt._phase_manager.flow_finished.connect(
            lambda success, msg: self._bus.emit(PhaseCompleteEvent(
                session_id=event.session_id,
                success=success,
                message=msg,
            ))
        )
        rt._phase_manager.error_occurred.connect(
            lambda code, detail: self._bus.emit(PhaseErrorEvent(
                session_id=event.session_id,
                code=code,
                detail=detail,
            ))
        )
    
    # 3. 启动 Phase
    task = rt.queue.get_streaming_task()
    if task:
        rt._phase_manager.start(task.user_text, task.mode, self._build_context(event.session_id))
    
    # 4. 更新任务状态
    self._repo.update_task_state(TaskState(
        session_id=event.session_id,
        phase=TaskPhase.ANALYZING,
    ))

def _on_phase_execute(self, event):
    """Execute 阶段：通知 Worker 执行任务"""
    rt = self._require_runtime(event.session_id)
    if not rt or not rt.worker:
        return
    
    # 调用 Worker 的 request_execute（如果 Worker 有该方法）
    if hasattr(rt.worker, "request_execute"):
        rt.worker.request_execute(
            event.task_list,
            event.original_text,
            self._build_context(event.session_id),
        )

def _on_phase_verify(self, event):
    """Verify 阶段：通知 Worker 执行验证"""
    rt = self._require_runtime(event.session_id)
    if not rt or not rt.worker:
        return
    
    if hasattr(rt.worker, "request_verify"):
        rt.worker.request_verify(
            event.execution_results,
            "",  # local_details
            self._build_context(event.session_id),
        )
```

class MainWindow(QMainWindow):
    def __init__(self):
        # ... 现有代码 ...

        from services.context_service import ContextService
        from services.project_service import ProjectService
        
        self._context_service = ContextService(...)
        self._project_service = ProjectService(...)
    
    def _get_project_path(self) -> str:
        """获取当前项目路径（从现有 ContextService）"""
        return self._context_service.get_project_root() or ""
    
    def _on_send_message(self, user_text: str):
        """发送消息"""
        current_sid = self._orchestrator.current_session_id
        if not current_sid:
            return
        
        # 接入现有模型选择器
        current_model = self._get_current_model()
        current_mode = self._get_current_mode()
        
        self.chat_view.clear_input()
        self._bus.emit(UserSendEvent(
            session_id=current_sid,
            user_text=user_text,
            mode=current_mode,
        ))
    
    def _get_current_model(self) -> str:
        """获取当前模型名称"""
        # 从 ChatView 的 model_selector 获取
        if hasattr(self.chat_view, "model_selector"):
            idx = self.chat_view.model_selector.currentIndex()
            if idx >= 0:
                return self.chat_view.model_selector.itemData(idx) or "tool-agent"
        return "tool-agent"
    
    def _get_current_mode(self) -> str:
        """获取当前模式"""
        # 从 ChatView 的 mode_group 获取
        if hasattr(self.chat_view, "mode_group"):
            for btn in self.chat_view.mode_group.buttons():
                if btn.isChecked():
                    return btn.property("mode") or "ask"
        return "ask"

## 适配 4：旧数据迁移

### 文件：`v4/repository.py`

### 需要适配的代码

```python
class SessionRepository:
    def __init__(self, db_path=DB_PATH):
        self._db_path = db_path
        os.makedirs(os.path.dirname(db_path), exist_ok=True)
        self._init_db()
        self._migrate_from_v3()
    
    def _migrate_from_v3(self):
        """从 v3 数据库迁移数据（可选）"""
        v3_db = "storage/conversations.db"
        if not os.path.exists(v3_db):
            return
        
        # 检查是否已经迁移
        with self._connect() as conn:
            count = conn.execute("SELECT COUNT(*) FROM sessions").fetchone()[0]
            if count > 0:
                return  # 已有数据，不迁移
        
        # 迁移会话
        v3_conn = sqlite3.connect(v3_db)
        v3_conn.row_factory = sqlite3.Row
        try:
            # 迁移会话
            sessions = v3_conn.execute(
                "SELECT id, title, mode, model, created_at, updated_at, project_path FROM conversations"
            ).fetchall()
            
            for row in sessions:
                self.create_session(SessionMetadata(
                    session_id=row["id"],
                    title=row["title"] or "新对话",
                    session_type=SessionType.WORK if row["project_path"] else SessionType.CHAT,
                    mode=row["mode"] or "ask",
                    model=row["model"] or "tool-agent",
                    project_path=row["project_path"] or "",
                    created_at=datetime.fromisoformat(row["created_at"]) if row["created_at"] else datetime.now(),
                    updated_at=datetime.fromisoformat(row["updated_at"]) if row["updated_at"] else datetime.now(),
                ))
            
            # 迁移消息
            messages = v3_conn.execute(
                "SELECT conversation_id, role, content, tool_calls, created_at FROM messages"
            ).fetchall()
            
            for row in messages:
                self.add_message(Message(
                    message_id=f"msg_{uuid.uuid4().hex[:16]}",
                    session_id=row["conversation_id"],
                    role=row["role"],
                    content=row["content"] or "",
                    tool_calls=row["tool_calls"],
                    created_at=datetime.fromisoformat(row["created_at"]) if row["created_at"] else datetime.now(),
                ))
        finally:
            v3_conn.close()
```

## 常见问题（FAQ）

### Q1：v4 是否兼容现有的 PyInstaller 打包？

**A**：v4 是 Python 模块，不引入新的系统依赖，兼容现有打包流程。只需确保 `v4/` 目录被打包工具包含。

### Q2：v4 是否支持多线程安全？

**A**：v4 的 `QueueManager` 使用 `threading.RLock`，`WorkerManager` 的 `_workers` 字典操作也在锁内。`MessageBus` 使用 `Qt.QueuedConnection`，天然线程安全。

### Q3：如果 Worker 在后台崩溃，UI 如何感知？

**A**：`WorkerManager` 桥接 `error_occurred` 信号到 `WorkerErrorEvent`，即使后台会话也会收到事件。`UIRenderer` 会丢弃非当前会话的 `WorkerErrorEvent`，但 `SessionOrchestrator` 会处理所有 `WorkerErrorEvent` 并更新任务状态。切回会话时，状态徽章显示 `failed`。

### Q4：Phase 的 Confirm 阶段跨会话如何处理？

**A**：`PhaseManager` 的 `confirm_required` 信号发射 `PhaseConfirmEvent`，`UIRenderer` 只处理当前会话的 `UIShowConfirmEvent`。如果用户在 Confirm 阶段切到另一个会话，Confirm 卡片隐藏。切回时，`SessionOrchestrator._switch_session` 从 `SessionRuntime.phase_state` 恢复 Confirm 状态，`UIRenderer` 重新显示 Confirm 卡片。

### Q5：并发槽位 5 是否可调？

**A**：`WorkerManager.MAX_WORKERS = 5` 是类常量，可在创建时传入：
```python
wm = WorkerManager(bus, max_workers=10)  # 调为 10
```
