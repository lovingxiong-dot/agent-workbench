```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List
from enum import Enum
import uuid

class SessionType(Enum):
    CHAT = "chat"
    WORK = "work"

class TaskPhase(Enum):
    IDLE = "idle"
    ANALYZING = "analyzing"
    CONFIRMING = "confirming"
    EXECUTING = "executing"
    VERIFYING = "verifying"
    ARCHIVING = "archiving"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"

@dataclass(frozen=True)
class SessionMetadata:
    session_id: str
    title: str
    session_type: SessionType
    mode: str
    model: str
    project_path: str = ""
    pinned: bool = False
    updated_at: datetime = field(default_factory=datetime.now)
    created_at: datetime = field(default_factory=datetime.now)

    @classmethod
    def new(cls, title, session_type, mode, model, project_path=""):
        return cls(
            session_id=f"sess_{uuid.uuid4().hex[:16]}",
            title=title,
            session_type=SessionType(session_type) if isinstance(session_type, str) else session_type,
            mode=mode, model=model, project_path=project_path,
        )

@dataclass(frozen=True)
class Message:
    message_id: str
    session_id: str
    role: str
    content: str
    tool_calls: Optional[str] = None
    created_at: datetime = field(default_factory=datetime.now)

    @classmethod
    def new(cls, session_id, role, content):
        return cls(
            message_id=f"msg_{uuid.uuid4().hex[:16]}",
            session_id=session_id, role=role, content=content,
        )

@dataclass(frozen=True)
class Environment:
    project_root: str
    tools: List[str] = field(default_factory=list)
    interpreter: str = ""
    workspace_context: str = ""

@dataclass(frozen=True)
class TaskState:
    session_id: str
    phase: TaskPhase
    task_count: int = 0
    current_task: int = 0
    updated_at: datetime = field(default_factory=datetime.now)
```

def _switch_session(self, new_session_id):
    # 核心：不操作 Worker，只切 UI
    self._current_session_id = new_session_id

def _create_worker(self, event):
    from workers.agent_worker import AgentWorker
    
    # 解析 LLM（接入现有 LLMRegistry）
    llm = self._resolve_llm(event.model)
    if not llm:
        self._bus.emit(WorkerErrorEvent(
            session_id=event.session_id,
            code="LLM_NOT_FOUND",
            detail=f"Model {event.model} not found",
        ))
        return
    
    worker = AgentWorker(
        mode_name=event.mode,
        current_llm=llm,
        session_id=event.session_id,
        project_root=event.project_root,
        # 从现有 main_window.py 复制其他参数
        system_prompt=self._build_system_prompt(event.project_root),
        tool_map=self._get_tool_map(),
        tool_definitions=self._get_tool_definitions(),
    )
    
    # 桥接信号（复制现有 main_window.py 中的连接逻辑）
    worker.chunk_ready.connect(
        lambda chunk: self._bus.emit(WorkerChunkEvent(
            session_id=event.session_id,
            worker_id=worker.worker_id,
            chunk=chunk,
        ))
    )
    # ... 其他信号桥接
    
    worker.start()

def _on_queue_task_ready(self, event):
    rt = self._require_runtime(event.session_id)
    if not rt:
        return
    
    # 获取现有 PhaseManager（从 SessionRuntime 或外部传入）
    # 如果 PhaseManager 在 SessionRuntime 中：
    # phase_manager = rt.phase_manager
    # 如果 PhaseManager 是全局单例：
    from agent_engine.phase_manager import PhaseManager
    phase_manager = PhaseManager(parent=self)
    
    # 启动 Phase 流程
    task = rt.queue.get_streaming_task()
    if task:
        phase_manager.start(task.user_text, task.mode, self._build_context(event.session_id))
    
    # 连接 PhaseManager 信号到事件总线
    phase_manager.phase_changed.connect(
        lambda phase, mode: self._bus.emit(PhaseChangedEvent(
            session_id=event.session_id,
            phase=phase,
            task_count=0,
        ))
    )
    # ... 其他信号连接
